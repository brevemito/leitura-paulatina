<script>
var S=window.localStorage,cat=null,Q={};
function $(i){return document.getElementById(i)}
function g(k,d){try{var v=S.getItem(k);return v?JSON.parse(v):d}catch(e){return d}}
function s(k,v){try{S.setItem(k,JSON.stringify(v))}catch(e){}}
function esc(t){return t.replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;")}
function bold(x){return x.replace(/\*\*(.+?)\*\*/g,"<strong>$1</strong>")}
function md(t){
  return esc(t).trim().split(/\n\s*\n/).map(function(b){
    var L=b.split("\n");
    if(/^- /.test(L[0]))return "<ul>"+L.map(function(x){return "<li>"+bold(x.replace(/^- /,""))+"</li>"}).join("")+"</ul>";
    return "<p>"+bold(L.join(" "))+"</p>";
  }).join("");
}
function parse(t){
  var o={},p=t.split(/^## /m);
  o.titulo=(p[0].match(/^# (.+)$/m)||[0,""])[1];
  for(var i=1;i<p.length;i++){var n=p[i].indexOf("\n");o[p[i].slice(0,n).trim()]=p[i].slice(n+1)}
  return o;
}
function quiz(txt,id){
  Q={};
  return txt.trim().split(/\n\s*\n/).map(function(b,n){
    var L=b.split("\n"),q=L[0].replace(/^\d+\.\s*/,""),ops=[],r="",ex="",k="q:"+id+":"+n;
    L.slice(1).forEach(function(x){
      if(/^[a-d]\) /.test(x))ops.push(x);
      else if(/^Resposta:/.test(x))r=x.replace(/^Resposta:\s*/,"").trim();
      else if(/^Explicação:/.test(x))ex=x.replace(/^Explicação:\s*/,"");
    });
    Q[k]={r:r,ex:ex};
    return "<div class=\"q\" data-k=\""+k+"\"><p><strong>"+(n+1)+". "+esc(q)+"</strong></p>"+
      ops.map(function(o){return "<label><input type=\"radio\" name=\""+k+"\" value=\""+o.charAt(0)+"\"> "+esc(o.slice(3))+"</label>"}).join("")+
      "<button>Verificar</button><div class=\"fb\"></div></div>";
  }).join("");
}
function mark(d,v){
  var q=Q[d.dataset.k],ok=(v===q.r);
  d.querySelector(".fb").innerHTML=(ok?"✓ Correcto. ":"✗ Não é esta. Resposta certa: "+q.r+". ")+esc(q.ex);
  d.className="q "+(ok?"ok":"no");
}
function sc(){
  $("sc").textContent="Pontuação: "+document.querySelectorAll(".q.ok").length+" de "+document.querySelectorAll(".q").length;
}
function show(b,f){
  var id=b+"/"+f;
  fetch("livros/"+b+"/"+f+".md").then(function(r){if(!r.ok)throw 0;return r.text()}).then(function(t){
    var o=parse(t);
    $("c").innerHTML="<h1>"+esc(o.titulo)+"</h1>"+
      "<h2>Texto original</h2><div class=\"tx\">"+md(o["Texto original"]||"")+"</div>"+
      "<h2>Vocabulário</h2>"+md(o["Vocabulário"]||"")+
      "<h2>Perguntas</h2>"+quiz(o["Perguntas"]||"",id)+"<p id=\"sc\"></p>"+
      "<h2>Resumo</h2>"+md(o["Resumo em uma frase"]||"")+
      "<nav><button id=\"bd\">Marcar como concluído</button></nav>";
    var saved=g("a",{});
    Array.prototype.forEach.call(document.querySelectorAll(".q"),function(d){
      var k=d.dataset.k;
      if(saved[k]){var i=d.querySelector("input[value=\""+saved[k]+"\"]");if(i)i.checked=true;mark(d,saved[k])}
      d.querySelector("button").onclick=function(){
        var c=d.querySelector("input:checked");
        if(!c){d.querySelector(".fb").textContent="Escolhe uma opção.";return}
        var a=g("a",{});a[k]=c.value;s("a",a);mark(d,c.value);sc();
      };
    });
    sc();
    if(g("done",{})[id])$("bd").textContent="Concluído ✓";
    $("bd").onclick=function(){var d=g("done",{});d[id]=1;s("done",d);$("bd").textContent="Concluído ✓"};
  }).catch(function(){$("c").innerHTML="<p>Trecho não encontrado.</p>"});
}
function fillT(){
  var b=cat.livros[$("lb").selectedIndex],d=g("done",{});
  $("lt").innerHTML=b.trechos.map(function(t){return "<option value=\""+t.id+"\">"+(d[b.id+"/"+t.id]?"✓ ":"")+t.id+" "+esc(t.titulo)+"</option>"}).join("");
  var f=b.trechos.filter(function(t){return !d[b.id+"/"+t.id]})[0]||b.trechos[0];
  $("lt").value=f.id;show(b.id,f.id);
}
function td(){return "t:"+new Date().toISOString().slice(0,10)}
function up(){$("tm").textContent="Hoje: "+Math.floor(g(td(),0)/60)+" min de leitura"}
setInterval(function(){if(!document.hidden){s(td(),g(td(),0)+10);up()}},10000);
up();
fetch("livros/catalogo.json").then(function(r){return r.json()}).then(function(c){
  cat=c;
  $("lb").innerHTML=c.livros.map(function(b){return "<option>"+esc(b.titulo)+"</option>"}).join("");
  $("lb").onchange=fillT;
  $("lt").onchange=function(){show(cat.livros[$("lb").selectedIndex].id,$("lt").value)};
  fillT();
}).catch(function(){$("c").innerHTML="<p>Não foi possível carregar o catálogo.</p>"});
</script>
